Python API access example: {{ project.name }}
import requests

USER_EMAIL = "{{ user.email }}"
# Don't share with anyone else
SECRET_TOKEN = "{{ user.token }}"

try:
    headers = {"email": USER_EMAIL, "token": SECRET_TOKEN}
    response = requests.get("{{ api_url }}", headers=headers)
    response.raise_for_status()
    data = response.json()
    print("Response JSON:")
    print(data)
except requests.exceptions.RequestException as e:
    print(f"Unable to access API: {e}")
except ValueError:
    print("Error parsing the response as JSON.")
